    SELECT cityname as city_name, 
           regionname as region_name       
    
    FROM {{ source('retail_databricks_bronze', 'api_cities') }}

        SELECT * 
        FROM source_data

        from deduplicated
    )

campaign_log.sql:

{{
    config(
    materialized = 'incremental',
    unique_key = 'campaign_log_key'
    )
}}

WITH source_data as (

    SELECT CampaignName, Channel, StartDate, EndDate, Budget, Date, Impressions, Clicks, Spend      
    
    FROM {{ source('retail_databricks_bronze', 'aws_s3_campaign_log') }}

    ),

    cleaned as (

        SELECT 
            INITCAP(TRIM(CampaignName)) as campaign_name,
            INITCAP(TRIM(Channel)) as channel,
            TO_DATE(TRIM(StartDate),"yyyy-MM-dd") as start_date,
            TO_DATE(TRIM(EndDate),"yyyy-MM-dd") as end_date,
            TRY_CAST(TRIM(Budget) AS DECIMAL(10, 2)) AS budget,
            TO_DATE(TRIM(Date),"yyyy-MM-dd") as campaign_date,
            TRY_CAST(TRIM(Impressions) AS INTEGER) AS impressions,
            TRY_CAST(TRIM(Clicks) AS INTEGER) AS clicks,
            TRY_CAST(TRIM(Spend) AS DECIMAL(10,2)) AS spend 
        
        FROM source_data

    ),


    deduplicated as (

        SELECT *
        FROM cleaned
        qualify
            row_number() over (
                partition by campaign_name,channel,start_date,end_date,budget,campaign_date,impressions,clicks,spend 
                order by campaign_name
            ) = 1

    ),

    add_key as (
        SELECT {{ generate_surrogate_key(['campaign_name', 'channel', 'start_date', 'end_date', 'campaign_date']) }} as campaign_log_key,*
        FROM deduplicated
    ),
    max_date as (
        select 
            coalesce(max(campaign_date),date('1900-01-01')) as max_campaign_date
        from add_key
    )


SELECT *,current_timestamp() as silver_process_at
FROM add_key


{% if is_incremental() %}
  where  campaign_date > (select  max_campaign_date from max_date)
{% endif %}

            month,

        FROM source_data

                `2025-01`,
                `2025-02`,

            TRY_CAST(NULLIF(TRIM(value), '') AS DECIMAL(18, 2)) AS inventory_quantity

dim_date.sql:


{{
    config(
        materialized='table'
    )
}}

WITH source_data as (

    SELECT MIN(min_date) as min_date,
           MAX(max_date) as max_date
    FROM {{ source('retail_databricks_gold', 'metadata_fact_date_ranges')}}
    

),
    buffered_date_range AS(
        SELECT
            ADD_MONTHS(
                min_date,
                -12
            ) AS start_date,

            ADD_MONTHS(
                max_date,
                12
            ) AS end_date
        FROM source_data

    ),
    dates AS(
        SELECT EXPLODE(SEQUENCE(start_date, end_date, INTERVAL 1 DAY)) as calendar_date
        FROM buffered_date_range
    ),
    calender_table AS(
        SELECT
               CAST(DATE_FORMAT(calendar_date,'yyyymmdd') AS INT) AS date_key,
               calendar_date,
               DAYOFWEEK(calendar_date) AS day_of_week,
               DAYOFYEAR(calendar_date) AS day_of_year,
               DATE_FORMAT(calendar_date,'EEEE') AS day_name,
               MONTH(calendar_date) AS MONTH,
               DATE_FORMAT(calendar_date,'MMMM') AS month_name,
               LAST_DAY(calendar_date) AS month_end_date,
               CONCAT(DATE_FORMAT(calendar_date,'MM'),'-',YEAR(calendar_date)) AS month_year,
               QUARTER(calendar_date) AS quarter_number,
               CONCAT('Q',QUARTER(calendar_date)) AS quarter_name,
               CONCAT('Q',QUARTER(calendar_date),'-',YEAR(calendar_date)) AS quarter_year,
               YEAR(calendar_date) AS year



        FROM dates
        
    )


SELECT *
       ,current_timestamp() as gold_process_at
FROM calender_table
